# Test and lint tooling.
#   pip install -r requirements.txt -r requirements-dev.txt

# The <9.0 cap held the suite on 8.x, which is affected by PYSEC-2026-1845;
# the fix is 9.0.3. Raising the floor past the advisory rather than pinning
# to it keeps pip-audit green without freezing the suite on one release.
pytest>=9.1.1,<10.0
pytest-asyncio>=0.23,<2.0
pytest-cov>=5.0,<8.0
# Was `pip install pytest-split` inline in ci.yml, unpinned, so a release could
# change how the suite divides without a commit here. Pinned like the rest.
pytest-split>=0.9,<1.0

# Parallel test execution. The suite was held back from this by an
# order-dependence: parameter_registry is a process-wide singleton and
# AutoTuningScheduler.start() left risk.ensemble_blend_weight registered for
# every test that followed it in the same worker. tests/conftest.py now
# snapshots and restores that registry per test, so worker assignment no
# longer changes results. This branch's earlier note saying xdist was
# deliberately excluded described the suite before that fix.
pytest-xdist>=3.6,<4.0
ruff==0.16.8
